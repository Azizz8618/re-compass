? .MOV *)
(*ДEЙCTBИE HAД CTOЛБЦAMИ*)
PROCEDURE MOV(CR:COLUMN;ZNAK:CHAR;SC:SET OF COLUMN);
VAR C:COLUMN;L:LINE;ПEPBЫЙ:BOOL;
BEGIN
IF PRINTCALC THEN_(ПEPBЫЙ:=TRUE;WCOL(CR);WRITE(OUT,' ',ZNAK)_);
FOR L:=MINLIN TO MAXLIN DO WITH T[CR,L].CON!.Д! DO_(
  FOR C:=FIRCOL TO LASCOL DO IF C IN SC THEN_(
    IF PRINTCALC AND (L=MINLIN) THEN_(
      IF ПEPBЫЙ THEN ПEPBЫЙ:=FALSE ELSE WRITE(OUT,' ',ZNAK);
      WCOL(C);WRITE(OUT,CLF)_);
    SELECT
      ZNAK='+':ЗH:=ЗH+T[C,L].CON!.Д!.ЗH;
      ZNAK='-':ЗH:=ЗH-T[C,L].CON!.Д!.ЗH_)_)_);
IF PRINTCALC THEN_(WRITE(OUT,' --> ');WCOL(CR);PUTP_);
END;



? .FILL *)
(*ЗAПOЛHEHИE ЯЧEЙKИ*)
PROCEDURE FILL(TT:TYPPAR;CC:COLUMN;LL:LINE;FF:CHAR);
VAR I:INT;
BEGIN
WITH T[CC,LL] DO_(
  NEW(CON);FRM:=FF;
  WITH CON! DO_(
    TYP:=TT;
    CASE TYP OF
      CEL,DAT,MES,KVA:_(NEW(Ц);Ц!.ЗH:=ЦEЛ.ЗH_);
      DEY:_(NEW(Д);Д!.ЗH:=ДEЙ.ЗH_);
      ALF:_(NEW(A);A!.ЗH:=AЛФ.ЗH_);
      STR:_(NEW(C);C!.ЗH:=CTP.ЗH_);
      LOG:_(NEW(Л);Л!.ЗH:=ЛOГ.ЗH_)_)_)_);
END;

? .DIV *)
(*PAЗБИEHИE TAБЛИЦЫ*)
PROCEDURE DIVIDE(FIRCOL,LASCOL:COLUMN);
VAR N:INT;C:COLUMN;
BEGIN
N:=1;C:=FIRCOL;
(W) WHILE C<LASCOL DO_(
  WITH INTERVAL[N] DO_(
    UP:=C;SIZ:=1;
    FOR C:=C TO LASCOL DO SIZ:=SIZ+FORCOL[C].S+1;
    C:=LASCOL;
    WHILE SIZ>SCREEN DO_(
      SIZ:=SIZ-FORCOL[C].S-1;C:=PRED(C)_);
    LOW:=C;
    IF C=LASCOL THEN EXIT W_);
  N:=N+1;C:=SUCC(C)_);
LASTINT:=N;
END;
